package com.example.diar;

import android.content.Intent;
import android.graphics.Color;
import android.net.Uri;
import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.util.Log;
import android.util.Patterns;
import android.widget.EditText;
import android.widget.ImageView;
import android.widget.Toast;

import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.appcompat.app.AppCompatActivity;

import com.google.android.material.button.MaterialButton;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class Interfacecreercompteetudiant extends AppCompatActivity {

    private EditText etNomComplet, etEmail, etPassword, etConfirmPassword;
    private MaterialButton btnCreerCompte;
    private ImageView imageView;

    private Uri selectedImageUri;
    private ActivityResultLauncher<String> galleryLauncher;

    private final ExecutorService executor = Executors.newSingleThreadExecutor();
    private final Handler handler = new Handler(Looper.getMainLooper());

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.interface_creer_compte_etudiant);

        // Association des composants
        etNomComplet = findViewById(R.id.etFullName);
        etEmail = findViewById(R.id.etEmail);
        etPassword = findViewById(R.id.etPassword);
        etConfirmPassword = findViewById(R.id.etConfirmPassword);
        btnCreerCompte = findViewById(R.id.btnSubmit);
        imageView = findViewById(R.id.imageView);

        // Ouvrir la galerie
        galleryLauncher = registerForActivityResult(
                new ActivityResultContracts.GetContent(),
                uri -> {
                    if (uri != null) {
                        selectedImageUri = uri;
                        imageView.setImageURI(uri);
                        imageView.setScaleType(ImageView.ScaleType.CENTER_CROP);
                    }
                }
        );

        // Cliquer sur l'image pour choisir une photo
        imageView.setOnClickListener(v ->
                galleryLauncher.launch("image/*")
        );

        // Bouton retour
        ImageView btnBack = findViewById(R.id.btnBack);
        // Le bouton retour est un ImageButton dans le XML
        if (btnBack != null) {
            btnBack.setOnClickListener(v -> finish());
        }

        btnCreerCompte.setOnClickListener(v -> validerInscription());
    }

    private void validerInscription() {

        String nom = etNomComplet.getText().toString().trim();
        String email = etEmail.getText().toString().trim();
        String password = etPassword.getText().toString().trim();
        String confirmPassword = etConfirmPassword.getText().toString().trim();

        resetFieldColors();

        // 1. Champs vides
        if (nom.isEmpty() && email.isEmpty() &&
                password.isEmpty() && confirmPassword.isEmpty()) {

            Toast.makeText(this, "Veuillez remplir tous les champs",
                    Toast.LENGTH_SHORT).show();

            setFieldError(etNomComplet);
            setFieldError(etEmail);
            setFieldError(etPassword);
            setFieldError(etConfirmPassword);
            return;
        }

        // 2. Nom complet
        if (nom.isEmpty()) {
            Toast.makeText(this, "Veuillez entrer votre nom complet",
                    Toast.LENGTH_SHORT).show();
            setFieldError(etNomComplet);
            etNomComplet.requestFocus();
            return;
        }

        int espaceCount = 0;

        for (int i = 0; i < nom.length(); i++) {
            char c = nom.charAt(i);

            if (c == ' ') {
                espaceCount++;
            } else if (!Character.isLetter(c)) {
                Toast.makeText(this,
                        "Le nom doit contenir uniquement des lettres",
                        Toast.LENGTH_LONG).show();
                setFieldError(etNomComplet);
                etNomComplet.requestFocus();
                return;
            }
        }

        if (espaceCount < 1 || espaceCount > 2) {
            Toast.makeText(this,
                    "Format incorrect ! (1 ou 2 espaces autorisés)",
                    Toast.LENGTH_LONG).show();
            setFieldError(etNomComplet);
            etNomComplet.requestFocus();
            return;
        }

        setFieldSuccess(etNomComplet);

        // 3. Email
        if (email.isEmpty()) {
            Toast.makeText(this, "Veuillez entrer votre e-mail",
                    Toast.LENGTH_SHORT).show();
            setFieldError(etEmail);
            etEmail.requestFocus();
            return;
        }

        if (!Patterns.EMAIL_ADDRESS.matcher(email).matches()) {
            Toast.makeText(this,
                    "Format d'e-mail invalide",
                    Toast.LENGTH_LONG).show();
            setFieldError(etEmail);
            etEmail.requestFocus();
            return;
        }

        setFieldSuccess(etEmail);

        // 4. Mot de passe
        if (password.isEmpty()) {
            Toast.makeText(this, "Veuillez entrer votre mot de passe",
                    Toast.LENGTH_SHORT).show();
            setFieldError(etPassword);
            etPassword.requestFocus();
            return;
        }

        if (password.length() < 6) {
            Toast.makeText(this,
                    "Le mot de passe doit contenir au moins 6 caractères",
                    Toast.LENGTH_LONG).show();
            setFieldError(etPassword);
            etPassword.requestFocus();
            return;
        }

        int chiffreCount = 0;

        for (int i = 0; i < password.length(); i++) {
            if (Character.isDigit(password.charAt(i))) {
                chiffreCount++;
            }
        }

        if (chiffreCount < 3) {
            Toast.makeText(this,
                    "Le mot de passe doit contenir au moins 3 chiffres",
                    Toast.LENGTH_LONG).show();
            setFieldError(etPassword);
            etPassword.requestFocus();
            return;
        }

        setFieldSuccess(etPassword);

        // 5. Confirmation
        if (confirmPassword.isEmpty()) {
            Toast.makeText(this,
                    "Veuillez confirmer votre mot de passe",
                    Toast.LENGTH_SHORT).show();
            setFieldError(etConfirmPassword);
            etConfirmPassword.requestFocus();
            return;
        }

        if (!password.equals(confirmPassword)) {
            Toast.makeText(this,
                    "Les mots de passe ne correspondent pas",
                    Toast.LENGTH_LONG).show();
            setFieldError(etConfirmPassword);
            etConfirmPassword.requestFocus();
            return;
        }

        setFieldSuccess(etConfirmPassword);

        // 6. Photo de profil
        if (selectedImageUri == null) {
            Toast.makeText(this,
                    "Veuillez choisir une photo de profil",
                    Toast.LENGTH_LONG).show();
            return;
        }

        // 7. Appel API
        btnCreerCompte.setEnabled(false);

        String emailClean = email.toLowerCase();

        executor.execute(() -> {

            boolean isSuccess = false;
            String serverMessage = "";

            HttpURLConnection conn = null;

            try {
                URL url = new URL(Config.BASE_URL + "/api/etudiants");


                conn = (HttpURLConnection) url.openConnection();

                conn.setRequestMethod("POST");
                conn.setRequestProperty(
                        "Content-Type", "application/json; charset=UTF-8"
                );
                conn.setRequestProperty("Accept", "application/json");
                conn.setConnectTimeout(15000);
                conn.setReadTimeout(15000);
                conn.setDoOutput(true);

                String jsonInputString =
                        "{\"nomComplet\":\"" + nom +
                                "\",\"email\":\"" + emailClean +
                                "\",\"motDePasse\":\"" + password + "\"}";

                byte[] input = jsonInputString.getBytes("UTF-8");

                conn.setFixedLengthStreamingMode(input.length);

                try (OutputStream os = conn.getOutputStream()) {
                    os.write(input);
                    os.flush();
                }

                int responseCode = conn.getResponseCode();

                BufferedReader br;

                if (responseCode >= 200 && responseCode < 300) {
                    isSuccess = true;
                    br = new BufferedReader(
                            new InputStreamReader(conn.getInputStream(), "UTF-8")
                    );
                } else {
                    br = new BufferedReader(
                            new InputStreamReader(conn.getErrorStream(), "UTF-8")
                    );
                }

                if (br != null) {
                    StringBuilder response = new StringBuilder();
                    String line;

                    while ((line = br.readLine()) != null) {
                        response.append(line);
                    }

                    serverMessage = response.toString();
                    br.close();
                }

                Log.d("InscriptionEtudiant",
                        "Code: " + responseCode + " | " + serverMessage);

            } catch (Exception e) {
                Log.e("InscriptionEtudiant", "Erreur API", e);
                serverMessage = e.getMessage() != null
                        ? e.getMessage() : "Erreur de connexion";
            } finally {
                if (conn != null) {
                    conn.disconnect();
                }
            }

            final boolean success = isSuccess;
            final String message = serverMessage;

            handler.post(() -> {

                btnCreerCompte.setEnabled(true);

                if (success) {

                    SessionManager sessionManager = new SessionManager(this);
                    sessionManager.sauvegarderSession(emailClean, nom);

                    Toast.makeText(this,
                            "Compte créé avec succès !",
                            Toast.LENGTH_LONG).show();

                    handler.postDelayed(() -> {

                        Intent intent = new Intent(
                                Interfacecreercompteetudiant.this,
                                Interface5.class
                        );

                        intent.putExtra("USER_EMAIL", emailClean);
                        intent.putExtra("USER_NAME", nom);

                        startActivity(intent);
                        finish();

                    }, 2500);

                } else {

                    Toast.makeText(this,
                            "Erreur : " + message,
                            Toast.LENGTH_LONG).show();
                }
            });
        });
    }

    // Couleur rouge en cas d'erreur
    private void setFieldError(EditText editText) {
        if (editText != null) {
            editText.setBackgroundColor(Color.parseColor("#FFCDD2"));
        }
    }

    // Couleur verte en cas de succès
    private void setFieldSuccess(EditText editText) {
        if (editText != null) {
            editText.setBackgroundColor(Color.parseColor("#C8E6C9"));
        }
    }

    // Réinitialisation des couleurs
    private void resetFieldColors() {
        setNeutralField(etNomComplet);
        setNeutralField(etEmail);
        setNeutralField(etPassword);
        setNeutralField(etConfirmPassword);
    }

    private void setNeutralField(EditText editText) {
        if (editText != null) {
            editText.setBackgroundColor(Color.TRANSPARENT);
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        executor.shutdown();
    }
}

package com.example.diar;

import android.content.Intent;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Color;
import android.net.Uri;
import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.util.Log;
import android.view.View;

import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.appcompat.app.AppCompatActivity;

import com.google.android.material.button.MaterialButton;
import com.google.android.material.card.MaterialCardView;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.io.OutputStreamWriter;

public class Interfaceverification extends AppCompatActivity {

    private MaterialCardView cardUploadBox;
    private ImageView ivPreview;
    private TextView btnBack; // Corrigé en TextView pour correspondre au XML
    private MaterialButton btnVerify;
    private String userEmail;
    private Uri documentUri = null;
    private File documentFileToSend = null;

    // pour la lecture et vérification de l'attestation
    private final ActivityResultLauncher<Intent> galleryLauncher = registerForActivityResult(
            new ActivityResultContracts.StartActivityForResult(),
            result -> {
                if (result.getResultCode() == RESULT_OK && result.getData() != null) {
                    documentUri = result.getData().getData();
                    if (documentUri != null) {
                        try {
                            // --- VÉRIFICATION DU TYPE DE FICHIER ---
                            String mimeType = getContentResolver().getType(documentUri);
                            if (mimeType == null || mimeType.equals("application/pdf") || !mimeType.startsWith("image/")) {
                                cardUploadBox.setCardBackgroundColor(Color.parseColor("#FFCDD2")); // Rouge clair
                                Toast.makeText(this, "Les fichiers PDF et non-images ne sont pas acceptés !", Toast.LENGTH_LONG).show();
                                documentUri = null;
                                documentFileToSend = null;
                                return; // On arrête le traitement ici
                            }
                            // ----------------------------------------

                            InputStream imageStream = getContentResolver().openInputStream(documentUri);
                            BitmapFactory.Options options = new BitmapFactory.Options();
                            options.inJustDecodeBounds = true;
                            BitmapFactory.decodeStream(imageStream, null, options);
                            imageStream.close();

                            int scaleFactor = 1;
                            int maxHeight = 1024;
                            int maxWidth = 1024;
                            if (options.outHeight > maxHeight || options.outWidth > maxWidth) {
                                final int heightRatio = Math.round((float) options.outHeight / (float) maxHeight);
                                final int widthRatio = Math.round((float) options.outWidth / (float) maxWidth);
                                scaleFactor = heightRatio < widthRatio ? heightRatio : widthRatio;
                            }

                            imageStream = getContentResolver().openInputStream(documentUri);
                            options.inJustDecodeBounds = false;
                            options.inSampleSize = scaleFactor;
                            Bitmap bitmap = BitmapFactory.decodeStream(imageStream, null, options);
                            imageStream.close();

                            ivPreview.setVisibility(View.VISIBLE);
                            ivPreview.setImageBitmap(bitmap);

                            documentFileToSend = new File(getCacheDir(), "attestation_temp.jpg");
                            FileOutputStream fos = new FileOutputStream(documentFileToSend);
                            bitmap.compress(Bitmap.CompressFormat.JPEG, 80, fos);
                            fos.flush();
                            fos.close();

                            cardUploadBox.setCardBackgroundColor(Color.parseColor("#C8E6C9")); // Vert clair si succès
                            Toast.makeText(this, "Attestation prête !", Toast.LENGTH_SHORT).show();

                        } catch (Exception e) {
                            e.printStackTrace();
                            cardUploadBox.setCardBackgroundColor(Color.parseColor("#FFCDD2")); // Rouge clair si erreur
                            Toast.makeText(this, "Erreur lors du traitement de l'image", Toast.LENGTH_SHORT).show();
                            documentFileToSend = null;
                        }
                    }
                }
            }
    );

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.interface_verif_etudiant);

        userEmail = getIntent().getStringExtra("USER_EMAIL");
        if (userEmail == null || userEmail.isEmpty()) {
            Toast.makeText(this, "Erreur : E-mail introuvable !", Toast.LENGTH_SHORT).show();
            finish();
            return;
        }

        cardUploadBox = findViewById(R.id.cardUploadBox);
        ivPreview = findViewById(R.id.ivPreview);
        btnVerify = findViewById(R.id.btnVerify);
        btnBack = findViewById(R.id.btnBack);

        if (btnBack != null) {
            btnBack.setOnClickListener(v -> finish());
        }

        cardUploadBox.setOnClickListener(v -> {
            Intent intent = new Intent(Intent.ACTION_PICK);
            intent.setType("image/*");
            galleryLauncher.launch(intent);
        });

        btnVerify.setOnClickListener(v -> {
            if (documentFileToSend == null || !documentFileToSend.exists()) {
                cardUploadBox.setCardBackgroundColor(Color.parseColor("#FFCDD2"));
                Toast.makeText(this, "Veuillez importer votre attestation", Toast.LENGTH_SHORT).show();
                return;
            }

            ExecutorService executor = Executors.newSingleThreadExecutor();
            Handler handler = new Handler(Looper.getMainLooper());

            executor.execute(() -> {
                final boolean[] isSuccess = {false};
                final String[] serverMessage = {""};

                try {
                    URL url = new URL(Config.BASE_URL + "/api/etudiants/verifier");
                    HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                    conn.setRequestMethod("POST");
                    conn.setDoOutput(true);

                    String boundary = "Boundary_" + System.currentTimeMillis();
                    conn.setRequestProperty("Content-Type", "multipart/form-data; boundary=" + boundary);

                    OutputStream outputStream = conn.getOutputStream();
                    PrintWriter writer = new PrintWriter(new OutputStreamWriter(outputStream, "UTF-8"), true);

                    writer.append("--").append(boundary).append("\r\n");
                    writer.append("Content-Disposition: form-data; name=\"email\"\r\n\r\n");
                    writer.append(userEmail).append("\r\n");

                    writer.append("--").append(boundary).append("\r\n");
                    writer.append("Content-Disposition: form-data; name=\"carte\"; filename=\"attestation.jpg\"\r\n");
                    writer.append("Content-Type: image/jpeg\r\n\r\n");
                    writer.flush();

                    try (InputStream fileInputStream = new FileInputStream(documentFileToSend)) {
                        byte[] buffer = new byte[4096];
                        int bytesRead;
                        while ((bytesRead = fileInputStream.read(buffer)) != -1) {
                            outputStream.write(buffer, 0, bytesRead);
                        }
                    }
                    outputStream.flush();
                    writer.append("\r\n");
                    writer.append("--").append(boundary).append("--\r\n");
                    writer.flush();

                    int responseCode = conn.getResponseCode();
                    if (responseCode == 200) {
                        isSuccess[0] = true;
                    } else {
                        if (conn.getErrorStream() != null) {
                            try (BufferedReader br = new BufferedReader(new InputStreamReader(conn.getErrorStream(), "utf-8"))) {
                                StringBuilder response = new StringBuilder();
                                String line;
                                while ((line = br.readLine()) != null) {
                                    response.append(line.trim());
                                }
                                serverMessage[0] = response.toString();
                            }
                        }
                    }
                } catch (Exception e) {
                    e.printStackTrace();
                    serverMessage[0] = "Exception : " + e.getMessage();
                }

                handler.post(() -> {
                    if (isSuccess[0]) {
                        Toast.makeText(this, "Vérification réussie avec succès !", Toast.LENGTH_LONG).show();

                        // Délai personnalisé de 2,5 secondes (2500 ms) avant redirection
                        handler.postDelayed(() -> {
                            Intent intent = new Intent(Interfaceverification.this, Interface6.class);
                            // Si vous avez besoin de passer l'email ou d'autres données à l'interface 6 :
                            // intent.putExtra("USER_EMAIL", userEmail);
                            startActivity(intent);
                            finish(); // Ferme l'activité actuelle pour ne pas y revenir en faisant "Retour"
                        }, 2500);

                    } else {
                        Toast.makeText(this, "Erreur de vérification : " + serverMessage[0], Toast.LENGTH_LONG).show();
                    }
                });
            });
        });
    }
}

    // Enregistrement du profil
    private void enregistrerProfil() {

        String genre = spinnerGenre.getText().toString().trim();
        String universite = spinnerUniversite.getText().toString().trim();
        String niveau = spinnerNiveau.getText().toString().trim();
        String faculte = spinnerFaculte.getText().toString().trim();

        if (genre.isEmpty() ||
                genre.equals("Sélectionnez le genre")) {

            spinnerGenre.setError("Choisissez votre genre");
            return;
        }

        if (universite.isEmpty() ||
                universite.equals("Sélectionnez l'université")) {

            spinnerUniversite.setError("Choisissez votre université");
            return;
        }

        if (niveau.isEmpty() ||
                niveau.equals("Sélectionnez le niveau")) {

            spinnerNiveau.setError("Choisissez votre niveau");
            return;
        }

        // Vérifier la faculté pour Université de Sousse
        if (universite.equals("Université de Sousse")) {

            if (faculte.isEmpty() ||
                    faculte.equals("Sélectionnez votre établissement")) {

                spinnerFaculte.setError(
                        "Choisissez votre faculté ou institut"
                );
                return;
            }
        } else {
            faculte = "";
        }

        if (userEmail == null || userEmail.trim().isEmpty()) {
            Toast.makeText(this,
                    "Email étudiant manquant",
                    Toast.LENGTH_LONG).show();
            return;
        }

        btnSuivant.setEnabled(false);
        btnSuivant.setText("Enregistrement...");

        final String faculteFinale = faculte;

        executor.execute(() -> {

            HttpURLConnection conn = null;
            String serverMessage = "";

            try {

                URL url = new URL(
                        BASE_URL + "/api/etudiants/profil"
                );

                conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("POST");
                conn.setDoOutput(true);
                conn.setConnectTimeout(15000);
                conn.setReadTimeout(15000);

                String boundary =
                        "Boundary_" + System.currentTimeMillis();

                conn.setRequestProperty(
                        "Content-Type",
                        "multipart/form-data; boundary=" + boundary
                );

                try (OutputStream outputStream = conn.getOutputStream();
                     PrintWriter writer = new PrintWriter(
                             new java.io.OutputStreamWriter(
                                     outputStream,
                                     StandardCharsets.UTF_8
                             ),
                             true
                     )) {

                    ajouterChamp(writer, boundary, "email", userEmail);
                    ajouterChamp(writer, boundary, "genre", genre);
                    ajouterChamp(writer, boundary, "universite", universite);
                    ajouterChamp(writer, boundary, "niveau", niveau);
                    ajouterChamp(writer, boundary, "faculte", faculteFinale);

                    writer.append("--")
                            .append(boundary)
                            .append("--\r\n");

                    writer.flush();
                }

                int responseCode = conn.getResponseCode();

                InputStream responseStream;

                if (responseCode >= 200 && responseCode < 300) {
                    responseStream = conn.getInputStream();
                } else {
                    responseStream = conn.getErrorStream();
                }

                if (responseStream != null) {

                    StringBuilder response = new StringBuilder();

                    try (BufferedReader reader = new BufferedReader(
                            new InputStreamReader(
                                    responseStream,
                                    StandardCharsets.UTF_8
                            ))) {

                        String line;

                        while ((line = reader.readLine()) != null) {
                            response.append(line);
                        }
                    }

                    serverMessage = response.toString();
                }

                final boolean success =
                        responseCode >= 200 && responseCode < 300;

                final String finalMessage = serverMessage;

                handler.post(() -> {

                    btnSuivant.setEnabled(true);
                    btnSuivant.setText("Suivant");

                    if (success) {

                        Toast.makeText(
                                Interface5.this,
                                "Profil enregistré avec succès !",
                                Toast.LENGTH_LONG
                        ).show();

                        Intent intent = new Intent(
                                Interface5.this,
                                Interface6.class
                        );

                        intent.putExtra("USER_EMAIL", userEmail);
                        intent.putExtra("USER_NAME", userName);

                        startActivity(intent);
                        finish();

                    } else {

                        Log.e("Interface5",
                                "Erreur serveur : " + finalMessage);

                        Toast.makeText(
                                Interface5.this,
                                "Erreur d'enregistrement : " + finalMessage,
                                Toast.LENGTH_LONG
                        ).show();
                    }
                });

            } catch (Exception e) {

                Log.e("Interface5", "Erreur connexion", e);

                handler.post(() -> {

                    btnSuivant.setEnabled(true);
                    btnSuivant.setText("Suivant");

                    Toast.makeText(
                            Interface5.this,
                            "Impossible de contacter le serveur : "
                                    + e.getMessage(),
                            Toast.LENGTH_LONG
                    ).show();
                });

            } finally {

                if (conn != null) {
                    conn.disconnect();
                }
            }
        });
    }

    // Ajouter un champ multipart
    private void ajouterChamp(
            PrintWriter writer,
            String boundary,
            String nom,
            String valeur
    ) throws Exception {

        writer.append("--")
                .append(boundary)
                .append("\r\n");

        writer.append(
                        "Content-Disposition: form-data; name=\""
                )
                .append(nom)
                .append("\"\r\n\r\n");

        writer.append(valeur != null ? valeur : "")
                .append("\r\n");

        writer.flush();
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        executor.shutdown();
        handler.removeCallbacksAndMessages(null);
    }
}

    // Enregistrement des préférences dans Spring Boot
    private void envoyerVersServeur(
            String email,
            String typeLogement,
            String location,
            double budget,
            String equipements,
            String quartier) {

        if (isSending) return;

        isSending = true;

        btnEnregistrer.setEnabled(false);
        btnEnregistrer.setText("Enregistrement...");

        executor.execute(() -> {

            HttpURLConnection conn = null;

            boolean success = false;
            String serverMessage = "";

            try {

                URL url = new URL(
                        BASE_URL + "/api/etudiants/preferences");

                conn = (HttpURLConnection)
                        url.openConnection();

                conn.setRequestMethod("POST");

                conn.setConnectTimeout(15000);
                conn.setReadTimeout(15000);

                conn.setDoOutput(true);

                conn.setRequestProperty(
                        "Content-Type",
                        "application/x-www-form-urlencoded; charset=UTF-8"
                );

                String params =
                        "email=" + URLEncoder.encode(email, "UTF-8") +
                                "&typeLogement=" + URLEncoder.encode(typeLogement, "UTF-8") +
                                "&locationType=" + URLEncoder.encode(location, "UTF-8") +
                                "&budgetMax=" + budget +
                                "&equipements=" + URLEncoder.encode(equipements, "UTF-8") +
                                "&quartier=" + URLEncoder.encode(quartier, "UTF-8");

                try (OutputStream os = conn.getOutputStream()) {

                    os.write(params.getBytes("UTF-8"));
                    os.flush();
                }

                int responseCode = conn.getResponseCode();

                InputStream responseStream;

                if (responseCode >= 200 && responseCode < 300) {

                    responseStream = conn.getInputStream();

                } else {

                    responseStream = conn.getErrorStream();
                }

                if (responseStream != null) {

                    StringBuilder response =
                            new StringBuilder();

                    try (BufferedReader reader =
                                 new BufferedReader(
                                         new InputStreamReader(
                                                 responseStream,
                                                 "UTF-8"
                                         ))) {

                        String line;

                        while ((line = reader.readLine()) != null) {
                            response.append(line);
                        }
                    }

                    serverMessage = response.toString();
                }

                success =
                        responseCode >= 200 &&
                                responseCode < 300;

            } catch (Exception e) {

                Log.e(
                        "Interface6",
                        "Erreur connexion serveur",
                        e
                );

                serverMessage = e.getMessage();

            } finally {

                if (conn != null) {
                    conn.disconnect();
                }
            }

            final boolean finalSuccess = success;
            final String finalMessage = serverMessage;

            handler.post(() -> {

                isSending = false;

                btnEnregistrer.setEnabled(true);
                btnEnregistrer.setText("Enregistrer");

                }
            });
        });

    private void chargerDonneesUtilisateur(String email) {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        Handler handler = new Handler(Looper.getMainLooper());

        final String[] nomUtilisateur = {"Utilisateur"};
        final String[] universiteUtilisateur = {"Université de Sousse"};
        final String[] photoUrl = {null}; // Initialisé à null proprement
        final boolean[] success = {false};

        executor.execute(() -> {
            try {
                // 1. Récupération du profil JSON
                URL url = new URL(Config.BASE_URL + "/api/etudiants/profil?email=" + URLEncoder.encode(email, "UTF-8"));
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("GET");
                conn.setRequestProperty("Accept", "application/json");

                int responseCode = conn.getResponseCode();
                if (responseCode == 200) {
                    BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream(), "utf-8"));
                    StringBuilder response = new StringBuilder();
                    String responseLine;
                    while ((responseLine = br.readLine()) != null) {
                        response.append(responseLine.trim());
                    }
                    br.close();

                    JSONObject jsonObject = new JSONObject(response.toString());

                    if (jsonObject.has("nomComplet") && !jsonObject.isNull("nomComplet")) {
                        nomUtilisateur[0] = jsonObject.getString("nomComplet");
                    }
                    if (jsonObject.has("universite") && !jsonObject.isNull("universite")) {
                        universiteUtilisateur[0] = jsonObject.getString("universite");
                    }
                    if (jsonObject.has("photoUrl") && !jsonObject.isNull("photoUrl")) {
                        photoUrl[0] = jsonObject.getString("photoUrl");
                        android.util.Log.d("DEBUG_PHOTO", "Nom du fichier trouvé en BDD : " + photoUrl[0]);
                    } else {
                        android.util.Log.d("DEBUG_PHOTO", "Le champ photoUrl est vide ou null dans le JSON !");
                    }

                    success[0] = true;
                } else {
                    android.util.Log.e("DEBUG_PHOTO", "Erreur HTTP : " + responseCode);
                }
            } catch (Exception e) {
                android.util.Log.e("DEBUG_PHOTO", "Erreur lors de l'appel profil : " + e.getMessage());
                e.printStackTrace();
            }

            // 2. Téléchargement de l'image si le nom du fichier existe
            Bitmap bitmapPhoto = null;
            if (success[0] && photoUrl[0] != null && !photoUrl[0].trim().isEmpty()) {
                try {
                    String imageUrl = Config.BASE_URL + "/uploads/" + photoUrl[0];
                    android.util.Log.d("DEBUG_PHOTO", "Tentative de téléchargement de l'image : " + imageUrl);

                    URL imageConnectionUrl = new URL(imageUrl);
                    InputStream inputStream = (InputStream) imageConnectionUrl.getContent();
                    bitmapPhoto = BitmapFactory.decodeStream(inputStream);

                    if (bitmapPhoto != null) {

    private void chargerProfilsCompatibles(String email) {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        Handler handler = new Handler(Looper.getMainLooper());

        executor.execute(() -> {
            try {
                // URL de ton endpoint Spring Boot
                URL url = new URL(Config.BASE_URL + "/api/etudiants/compatibles?email=" + email);
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("GET");
                conn.setRequestProperty("Accept", "application/json");

                int responseCode = conn.getResponseCode();
                if (responseCode == 200) {
                    BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream(), "utf-8"));
                    StringBuilder response = new StringBuilder();
                    String responseLine;
                    while ((responseLine = br.readLine()) != null) {
                        response.append(responseLine.trim());
                    }
                    br.close();

                    // Parsing du tableau JSON reçu du serveur (Format DTO : EtudiantAvecScore)
                    JSONArray jsonArray = new JSONArray(response.toString());
                    profilList.clear();

                    for (int i = 0; i < jsonArray.length(); i++) {
                        // 1. Récupération de l'objet global contenant le pourcentage et l'étudiant
                        JSONObject itemObj = jsonArray.getJSONObject(i);

                        // Récupération du pourcentage calculé par le backend
                        int pourcentage = itemObj.optInt("pourcentage", 50);

                        // 2. Extraction de l'objet imbriqué "etudiant"
                        JSONObject obj = itemObj.getJSONObject("etudiant");

                        // Récupération des champs de l'entité Etudiant
                        String nom = obj.optString("nomComplet", "Étudiant");
                        String userEmail = obj.optString("email", "");
                        String universite = obj.optString("universite", "Université de Sousse");
                        int age = obj.optInt("age", 21);
                        double budget = obj.optDouble("budgetMax", 300.0);
                        String quartier = obj.optString("quartier", "Non spécifié");
                        String modeDeVie = obj.optString("modeDeVie", "");
                        String nbColocataires = obj.optString("nbColocataires", "");
                        String proprete = obj.optString("proprete", "");
                        String animauxAcceptes = obj.optString("animauxAcceptes", "");
                        String photoUrl = obj.optString("photoUrl", ""); // <-- Récupération de l'URL de la photo

                        Etudiant etudiant = new Etudiant(nom, userEmail, universite, age, budget, modeDeVie, nbColocataires, proprete, animauxAcceptes);

                        // Injection des valeurs complémentaires dans l'objet Etudiant
                        etudiant.setPourcentage(pourcentage);
                        etudiant.setQuartier(quartier);
                        etudiant.setPhotoUrl(photoUrl); // <-- Assignation de la photo

                        profilList.add(etudiant);
                    }
                } else {
                    handler.post(() -> Toast.makeText(this, "Erreur serveur : " + responseCode, Toast.LENGTH_SHORT).show());
                }
            } catch (Exception e) {
                e.printStackTrace();
                handler.post(() -> Toast.makeText(this, "Erreur de connexion : " + e.getMessage(), Toast.LENGTH_SHORT).show());
            }

            // Mise à jour de l'UI sur le thread principal
            handler.post(() -> profilAdapter.notifyDataSetChanged());
        });
    }

    private void collecterEtEnvoyerDonnees() {
        // Récupération de l'e-mail de l'étudiant connecté transmis par l'Intent
        String email = getIntent().getStringExtra("USER_EMAIL");
        if (email == null || email.isEmpty()) {
            Toast.makeText(this, "Erreur : E-mail introuvable", Toast.LENGTH_SHORT).show();
            return;
        }

        // --- Récupération des choix multiples (Mode de vie) ---
        StringBuilder modeDeVieBuilder = new StringBuilder();
        for (int i = 0; i < chipGroupLifestyle.getChildCount(); i++) {
            if (chipGroupLifestyle.getChildAt(i) instanceof Chip) {
                Chip chip = (Chip) chipGroupLifestyle.getChildAt(i);
                if (chip.isChecked()) {
                    if (modeDeVieBuilder.length() > 0) {
                        modeDeVieBuilder.append(", ");
                    }
                    modeDeVieBuilder.append(chip.getText().toString());
                }
            }
        }
        String modeDeVie = modeDeVieBuilder.toString();

        // --- Récupération du nombre de colocataires acceptés (Sélection unique) ---
        String nbColocataires = "";
        int selectedRoommateId = chipGroupRoommates.getCheckedChipId();
        if (selectedRoommateId != android.view.View.NO_ID) {
            Chip selectedChip = findViewById(selectedRoommateId);
            if (selectedChip != null) {
                nbColocataires = selectedChip.getText().toString();
            }
        }

        // --- Récupération de la propreté (Sélection unique) ---
        String proprete = "";
        ChipGroup chipGroupProprete = findViewById(R.id.chipGroupProprete);
        int checkedIdProprete = chipGroupProprete.getCheckedChipId();
        if (checkedIdProprete != android.view.View.NO_ID) {
            Chip selectedChip = findViewById(checkedIdProprete);
            if (selectedChip != null) {
                proprete = selectedChip.getText().toString();
            }
        }

        // --- Récupération de l'acceptation des animaux (Sélection unique) ---
        String animauxAcceptes = "";
        int selectedPetId = chipGroupPets.getCheckedChipId();
        if (selectedPetId != android.view.View.NO_ID) {
            Chip selectedPetChip = findViewById(selectedPetId);
            if (selectedPetChip != null) {
                animauxAcceptes = selectedPetChip.getText().toString();
            }
        }

        // Validation simple : s'assurer que l'utilisateur a rempli au moins un champ
        if (modeDeVie.isEmpty() && nbColocataires.isEmpty() && proprete.isEmpty() && animauxAcceptes.isEmpty()) {
            Toast.makeText(this, "Veuillez sélectionner au moins une préférence.", Toast.LENGTH_SHORT).show();
            return;
        }

        // Envoi des données vers le serveur Spring Boot en arrière-plan (avec le paramètre proprete ajouté)
        envoyerDonneesAuServeur(email, modeDeVie, nbColocataires, proprete, animauxAcceptes);
    }

    private void envoyerDonneesAuServeur(String email, String modeDeVie, String nbColocataires, String proprete, String animauxAcceptes) {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        Handler handler = new Handler(Looper.getMainLooper());

        final boolean[] success = {false};
        final String[] messageReponse = {""};

        executor.execute(() -> {
            try {
                // URL de votre endpoint Spring Boot
                URL url = new URL(Config.BASE_URL + "/api/etudiants/profil-colocataire");
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("POST");
                conn.setRequestProperty("Content-Type", "application/json; utf-8");
                conn.setRequestProperty("Accept", "application/json");
                conn.setDoOutput(true);

                // Construction de l'objet JSON à envoyer
                JSONObject jsonParam = new JSONObject();
                jsonParam.put("email", email);
                jsonParam.put("modeDeVie", modeDeVie);
                jsonParam.put("nbColocataires", nbColocataires);
                jsonParam.put("proprete", proprete); // Ajouté ici pour correspondre au backend
                jsonParam.put("animauxAcceptes", animauxAcceptes);

                // Écriture des données dans la requête
                try (OutputStream os = conn.getOutputStream()) {
                    byte[] input = jsonParam.toString().getBytes("utf-8");
                    os.write(input, 0, input.length);
                }

                int responseCode = conn.getResponseCode();
                if (responseCode == 200 || responseCode == 201) {
                    BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream(), "utf-8"));
                    StringBuilder response = new StringBuilder();
                    String responseLine;
                    while ((responseLine = br.readLine()) != null) {
                        response.append(responseLine.trim());
                    }
                    br.close();
                    success[0] = true;
                    messageReponse[0] = "Critères enregistrés avec succès !";
                } else {
                    messageReponse[0] = "Erreur serveur (Code " + responseCode + ")";
                }
            } catch (Exception e) {
                e.printStackTrace();
                messageReponse[0] = "Erreur de connexion : " + e.getMessage();
            }

            // Retour sur le thread principal pour informer l'utilisateur
            handler.post(() -> {
                Toast.makeText(Interfaceprofilecoloc.this, messageReponse[0], Toast.LENGTH_LONG).show();
                if (success[0]) {
                    // Redirection vers l'Interface 7 (Accueil) après l'enregistrement du profil coloc
                    Intent intent = new Intent(Interfaceprofilecoloc.this, Interface7.class);
                    intent.putExtra("USER_EMAIL", email);
                    startActivity(intent);
                    finish();
                }
            });
        });
    }
}

            navProfil.setOnClickListener(v -> navigateTo(Interface17.class));
        }

    }

    protected void setupBackButton(Class<?> targetActivity) {
        View btnBack = findViewById(R.id.btnBack);

        if (btnBack != null) {
            btnBack.setOnClickListener(v -> {
                if (this.getClass().equals(targetActivity)) {
                    finish();
                } else {
                    navigateTo(targetActivity);
                }
            });
        }
    }

    /**
     * Méthode de navigation universelle et fluide
     */
    private void navigateTo(Class<?> targetActivity) {
        if (!this.getClass().equals(targetActivity)) {
            Intent intent = new Intent(this, targetActivity);
            // Évite d'empiler les mêmes activités en boucle et garde un historique propre
            intent.addFlags(Intent.FLAG_ACTIVITY_CLEAR_TOP | Intent.FLAG_ACTIVITY_SINGLE_TOP);
            startActivity(intent);
            overridePendingTransition(0, 0);
        }
    }
}

    // 1. Inscription initiale (Création de l'étudiant avec nom, email et mot de
    // passe)
    @PostMapping
    public ResponseEntity<?> creerEtudiant(@RequestBody Etudiant etudiant) {
        try {
            if (etudiant == null || etudiant.getEmail() == null || etudiant.getEmail().trim().isEmpty()) {
                return ResponseEntity.status(400).body("L'e-mail est obligatoire !");
            }

            String emailClean = etudiant.getEmail().trim();
            if (etudiantRepository.existsByEmail(emailClean)) {
                return ResponseEntity.status(409).body("Ce compte existe déjà !");
            }

            etudiant.setEmail(emailClean);

            // Sécurité anti-crash : Initialisation des valeurs par défaut pour éviter les
            // erreurs de contrainte en BDD
            if (etudiant.getStatutVerification() == null) {
                etudiant.setStatutVerification("NON_VERIFIE");
            }
            etudiant.setBudgetMax(0.0);

            Etudiant nouvelEtudiant = etudiantRepository.save(etudiant);
            return ResponseEntity.status(201).body(nouvelEtudiant);

        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body("Erreur serveur lors de la création : " + e.getMessage());
        }
    }

    // 4. Vérification automatique par OCR de l'attestation d'inscription
    @PostMapping("/verifier")
    public ResponseEntity<?> verifierEtudiant(
            @RequestParam("email") String email,
            @RequestParam("carte") MultipartFile fichierAttestation) {

        File tempFile = null;
        try {
            if (email != null)
                email = email.trim();
            Etudiant etudiant = etudiantRepository.findByEmail(email);
            if (etudiant == null) {
                List<Etudiant> tous = etudiantRepository.findAll();
                if (!tous.isEmpty()) {
                    etudiant = tous.get(0);
                } else {
                    return ResponseEntity.status(404).body("Étudiant introuvable avec cet e-mail.");
                }
            }

            tempFile = File.createTempFile("attestation_", ".jpg");
            Files.copy(fichierAttestation.getInputStream(), tempFile.toPath(), StandardCopyOption.REPLACE_EXISTING);

            Tesseract tesseract = new Tesseract();
            tesseract.setDatapath("C:/tessdata");
            tesseract.setLanguage("fra");

            String texteExtrait = tesseract.doOCR(tempFile);
            String texteMaj = texteExtrait.toUpperCase();

            System.out.println("=== TEXTE EXTRAIT PAR TESSERACT ===");
            System.out.println(texteMaj);
            System.out.println("==================================");

            // 1. VÉRIFICATION DES MOTS-CLÉS GLOBAUX (Séparée pour un message simple direct)
            boolean contientMotsCles = texteMaj.contains("ATTESTATION") ||
                    texteMaj.contains("INSCRIPTION") ||
                    texteMaj.contains("UNIVERSITE") ||
                    texteMaj.contains("INSTITUT") ||
                    texteMaj.contains("FACULTE");

            if (!contientMotsCles) {
                etudiant.setStatutVerification("NON_VERIFIE");
                etudiantRepository.save(etudiant);
                return ResponseEntity.status(400).body("Il ne s'agit pas d'une attestation d'inscription.");
            }

            // Liste pour accumuler toutes les autres erreurs de validation si c'est bien
            // une attestation
            List<String> erreurs = new java.util.ArrayList<>();

            // 2. VÉRIFICATION DE L'ANNÉE UNIVERSITAIRE
            LocalDate aujourdHui = LocalDate.now();
            int anneeActuelle = aujourdHui.getYear();
            int moisActuel = aujourdHui.getMonthValue();

            String anneeUniversitaire1, anneeUniversitaire2;
            if (moisActuel >= 9) {
                anneeUniversitaire1 = anneeActuelle + "-" + (anneeActuelle + 1);
                anneeUniversitaire2 = anneeActuelle + "/" + (anneeActuelle + 1);
            } else {
                anneeUniversitaire1 = (anneeActuelle - 1) + "-" + anneeActuelle;
                anneeUniversitaire2 = (anneeActuelle - 1) + "/" + anneeActuelle;
            }

            boolean anneePresente = texteMaj.contains(anneeUniversitaire1) || texteMaj.contains(anneeUniversitaire2);

            if (!anneePresente) {
                erreurs.add("L'attestation n'est pas valide pour l'année universitaire en cours (" + anneeUniversitaire1
                        + ").");
            }

            // 3. VÉRIFICATION DU NOM COMPLET (Sans Regex)
            String nomEtudiant = etudiant.getNomComplet() != null ? etudiant.getNomComplet().toUpperCase().trim() : "";
            boolean nomPresent = false;

            if (!nomEtudiant.isEmpty()) {
                String[] motsNom = nomEtudiant.split(" ");
                int motsTrouves = 0;
                int totalMots = 0;

                for (String mot : motsNom) {
                    mot = mot.trim();
                    if (mot.length() >= 3) {
                        totalMots++;
                        if (texteMaj.contains(mot)) {
                            motsTrouves++;
                        }
                    }
                }

                if (totalMots == 0 || motsTrouves >= 1) {
                    nomPresent = true;
                }
            } else {
                nomPresent = true;
            }

            if (!nomPresent) {
                erreurs.add("Le nom complet ne correspond pas au document.");
            }

            // 4. VÉRIFICATION STRICTE DU NIVEAU D'ÉTUDE (Sans Regex)
            String niveauEtudiant = etudiant.getNiveau() != null ? etudiant.getNiveau().toUpperCase().trim() : "";
            boolean niveauPresent = true;

            if (!niveauEtudiant.isEmpty()) {
                String[] motsNiveau = niveauEtudiant.split(" ");
                int motsTotal = 0;
                int motsTrouves = 0;

                for (String mot : motsNiveau) {
                    mot = mot.trim();
                    // On ignore les mots de liaison trop courts
                    if (mot.length() >= 2 && !mot.equals("DE") && !mot.equals("LA") && !mot.equals("ET")
                            && !mot.equals("EN") && !mot.equals("DU")) {
                        motsTotal++;
                        if (texteMaj.contains(mot)) {
                            motsTrouves++;
                        }
                    }
                }

                // Pour que le niveau soit valide, TOUS les mots significatifs (ex: le chiffre
                // et "année")
                // doivent être présents dans le texte extrait par l'OCR.
                if (motsTotal > 0 && motsTrouves < motsTotal) {
                    niveauPresent = false;
                }
            }

            if (!niveauPresent) {
                erreurs.add("Le niveau d'étude ne correspond pas au document.");
            }
            // 5. VÉRIFICATION DE L'UNIVERSITÉ (Sans Regex)
            String universiteEtudiante = etudiant.getUniversite() != null
                    ? etudiant.getUniversite().toUpperCase().trim()
                    : "";
            boolean universitePresente = true;

            if (!universiteEtudiante.isEmpty()) {
                String[] motsUniv = universiteEtudiante.split(" ");
                boolean trouveUniv = false;
                for (String mot : motsUniv) {
                    mot = mot.trim();
                    if (mot.length() > 3 && texteMaj.contains(mot)) {
                        trouveUniv = true;
                        break;
                    }
                }
                universitePresente = trouveUniv;
            }

            if (!universitePresente) {
                erreurs.add("L'université ne correspond pas au document.");
            }

            // S'il y a des erreurs, on les retourne toutes ensemble
            if (!erreurs.isEmpty()) {
                etudiant.setStatutVerification("NON_VERIFIE");
                etudiantRepository.save(etudiant);
                return ResponseEntity.status(400).body("Échec de vérification : " + String.join(" | ", erreurs));
            }

            // 6. VALIDATION FINALE RÉUSSIE
            etudiant.setStatutVerification("VERIFIE");
            etudiantRepository.save(etudiant);
            return ResponseEntity.ok("Vérification automatique réussie ! Compte validé.");

        } catch (TesseractException e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body("Erreur lors de l'analyse OCR : " + e.getMessage());
        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body("Erreur serveur : " + e.getMessage());
        } finally {
            if (tempFile != null && tempFile.exists()) {
                tempFile.delete();
            }
        }
    }

        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body("Erreur serveur : " + e.getMessage());
        }
    }

    // 6. Enregistrement du profil colocataire
    @PostMapping("/profil-colocataire")
    public ResponseEntity<?> enregistrerProfilColocataire(@RequestBody Map<String, String> requestBody) {
        try {
            String email = requestBody.get("email");
            String modeDeVie = requestBody.get("modeDeVie");
            String nbColocataires = requestBody.get("nbColocataires");
            String animauxAcceptes = requestBody.get("animauxAcceptes");

            // 🆕 Récupération du nouveau critère propreté
            String proprete = requestBody.get("proprete");

            if (email != null) {
                email = email.trim();
            }

            Etudiant etudiant = null;
            if (email != null && !email.isEmpty()) {
                etudiant = etudiantRepository.findByEmail(email);
            }

            if (etudiant == null) {
                List<Etudiant> tousLesEtudiants = etudiantRepository.findAll();
                if (!tousLesEtudiants.isEmpty()) {
                    etudiant = tousLesEtudiants.get(0);
                }
            }

            if (etudiant == null) {
                return ResponseEntity.status(404).body("Aucun étudiant trouvé dans la base de données.");
            }

            // Mise à jour de tous les champs (y compris la propreté)
            etudiant.setModeDeVie(modeDeVie);
            etudiant.setNbColocataires(nbColocataires);
            etudiant.setAnimauxAcceptes(animauxAcceptes);
            etudiant.setProprete(proprete); // 🆕 Enregistrement

            etudiantRepository.save(etudiant);

            return ResponseEntity.ok("Profil colocataire enregistré avec succès !");
        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body("Erreur serveur : " + e.getMessage());
        }
    }

    @GetMapping("/compatibles")
    public ResponseEntity<List<EtudiantAvecScore>> obtenirProfilsCompatibles(@RequestParam("email") String email) {
        System.out.println(">>> Email reçu depuis Android : [" + email + "]");
        try {
            Etudiant etudiantConnecte = etudiantRepository.findByEmail(email);
            if (etudiantConnecte == null) {
                return ResponseEntity.status(404).body(null);
            }

            List<Etudiant> tousLesEtudiants = etudiantRepository.findAll();
            List<EtudiantAvecScore> resultats = new ArrayList<>();

            String genreConnecte = etudiantConnecte.getGenre(); // "Femme" ou "Homme"

            for (Etudiant autre : tousLesEtudiants) {
                // 1. Ne pas se comparer soi-même
                if (autre.getEmail() != null && autre.getEmail().equalsIgnoreCase(email)) {
                    continue;
                }

                // 2. FILTRE STRICT PAR GENRE : Si l'étudiant connecté est une femme, afficher
                // uniquement les femmes (et vice-versa)
                if (genreConnecte != null && !genreConnecte.equalsIgnoreCase(autre.getGenre())) {
                    continue; // Ignore ce profil s'il ne respecte pas la règle de genre
                }

                int score = 20; // Score de base minimum

                // 3. Université identique (+25 points)
                if (etudiantConnecte.getUniversite() != null &&
                        etudiantConnecte.getUniversite().equalsIgnoreCase(autre.getUniversite())) {
                    score += 25;
                }

                // 4. Quartier identique (+20 points)
                if (etudiantConnecte.getQuartier() != null &&
                        etudiantConnecte.getQuartier().equalsIgnoreCase(autre.getQuartier())) {
                    score += 20;
                }

                // 5. Type de logement identique (+15 points)
                if (etudiantConnecte.getTypeLogement() != null &&
                        etudiantConnecte.getTypeLogement().equalsIgnoreCase(autre.getTypeLogement())) {
                    score += 15;
                }

                // 6. Budget max proche (écart de 50 DT max -> +10 points)
                if (Math.abs(autre.getBudgetMax() - etudiantConnecte.getBudgetMax()) <= 50) {
                    score += 10;
                }

                // 7. Propreté (+10 points)
                if (etudiantConnecte.getProprete() != null &&
                        etudiantConnecte.getProprete().equalsIgnoreCase(autre.getProprete())) {
                    score += 10;
                }

                // Plafonner le score à 100%
                if (score > 100)
                    score = 100;

                // Ajouter le résultat dans la liste
                resultats.add(new EtudiantAvecScore(autre, score));
            }

            // Trier par pourcentage décroissant (du plus compatible au moins compatible)
            resultats.sort((a, b) -> Integer.compare(b.getPourcentage(), a.getPourcentage()));

            return ResponseEntity.ok(resultats);
        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(500).body(null);
        }
    }
    // 7. Enregistrement de la photo de profil lors de l'inscription
@PostMapping(value = "/photo", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
public ResponseEntity<?> enregistrerPhotoInscription(
        @RequestParam("email") String email,
        @RequestParam("photo") MultipartFile photo) {

    try {
        if (email == null || email.trim().isEmpty()) {
            return ResponseEntity.badRequest().body("Email obligatoire !");
        }

        if (photo == null || photo.isEmpty()) {
            return ResponseEntity.badRequest().body("Photo obligatoire !");
        }

        if (photo.getSize() > 5 * 1024 * 1024) {
            return ResponseEntity.badRequest()
                    .body("Photo trop volumineuse (5 MB maximum).");
        }

        String contentType = photo.getContentType();
        String extension;

        if ("image/jpeg".equals(contentType)) {
            extension = ".jpg";
        } else if ("image/png".equals(contentType)) {
            extension = ".png";
        } else if ("image/webp".equals(contentType)) {
            extension = ".webp";
        } else {
            return ResponseEntity.badRequest()
                    .body("Format photo non accepté !");
        }

        String emailClean = email.trim();

        Etudiant etudiant = etudiantRepository.findByEmail(emailClean);

        if (etudiant == null) {
            return ResponseEntity.status(404)
                    .body("Étudiant introuvable !");
        }

        Path dossier = Paths.get("uploads", "etudiants");
        Files.createDirectories(dossier);

        String nomFichier = UUID.randomUUID().toString() + extension;

        Path destination = dossier.resolve(nomFichier);

        try (java.io.InputStream input = photo.getInputStream()) {
            Files.copy(input, destination,
                    StandardCopyOption.REPLACE_EXISTING);
        }

        etudiant.setPhotoUrl("/uploads/etudiants/" + nomFichier);

        etudiantRepository.save(etudiant);

        Map<String, String> resultat = new HashMap<>();
        resultat.put("message", "Photo enregistrée avec succès !");
        resultat.put("photoUrl", etudiant.getPhotoUrl());

        return ResponseEntity.ok(resultat);

    } catch (Exception e) {
        e.printStackTrace();

        return ResponseEntity.status(500)
                .body("Erreur lors de l'enregistrement de la photo : "
                        + e.getMessage());
    }
}
